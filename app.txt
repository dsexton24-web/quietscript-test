// QuietScript audio engine.
// Takes ordinary speech audio, band-limits it, and shifts it up into the
// near-ultrasonic range using single-sideband (SSB) modulation.
// Also verifies the result and can shift it back down (Reveal).
// Pure JavaScript, no dependencies, runs in a browser, a Web Worker or Node.

const OUTPUT_RATE = 48000;

// ---------- small helpers ----------

function peak(x) {
  let p = 0;
  for (let i = 0; i < x.length; i++) {
    const a = Math.abs(x[i]);
    if (a > p) p = a;
  }
  return p;
}

function scaleTo(x, target) {
  const p = peak(x);
  if (p === 0) return x;
  const g = target / p;
  for (let i = 0; i < x.length; i++) x[i] *= g;
  return x;
}

function nextPow2(n) {
  let p = 1;
  while (p < n) p <<= 1;
  return p;
}

// Linear-interpolation resampler. Good enough here because the signal is
// band-limited to a few kHz right after this step.
function resampleLinear(input, inRate, outRate) {
  if (inRate === outRate) return Float32Array.from(input);
  const ratio = inRate / outRate;
  const outLen = Math.max(1, Math.floor(input.length / ratio));
  const out = new Float32Array(outLen);
  for (let i = 0; i < outLen; i++) {
    const pos = i * ratio;
    const i0 = Math.floor(pos);
    const i1 = Math.min(i0 + 1, input.length - 1);
    const f = pos - i0;
    out[i] = input[i0] * (1 - f) + input[i1] * f;
  }
  return out;
}

// ---------- FFT (iterative radix-2, in place) ----------

function fft(re, im, inverse = false) {
  const n = re.length;
  for (let i = 1, j = 0; i < n; i++) {
    let bit = n >> 1;
    for (; j & bit; bit >>= 1) j ^= bit;
    j ^= bit;
    if (i < j) {
      let t = re[i]; re[i] = re[j]; re[j] = t;
      t = im[i]; im[i] = im[j]; im[j] = t;
    }
  }
  for (let len = 2; len <= n; len <<= 1) {
    const ang = (2 * Math.PI / len) * (inverse ? 1 : -1);
    const wr = Math.cos(ang), wi = Math.sin(ang);
    const half = len >> 1;
    for (let i = 0; i < n; i += len) {
      let cr = 1, ci = 0;
      for (let k = 0; k < half; k++) {
        const a = i + k, b = a + half;
        const xr = re[b] * cr - im[b] * ci;
        const xi = re[b] * ci + im[b] * cr;
        re[b] = re[a] - xr; im[b] = im[a] - xi;
        re[a] += xr; im[a] += xi;
        const ncr = cr * wr - ci * wi;
        ci = cr * wi + ci * wr;
        cr = ncr;
      }
    }
  }
  if (inverse) {
    for (let i = 0; i < n; i++) { re[i] /= n; im[i] /= n; }
  }
}

// ---------- FIR filter design (windowed sinc, Blackman window) ----------

function blackman(n, M) {
  return 0.42 - 0.5 * Math.cos((2 * Math.PI * n) / (M - 1)) + 0.08 * Math.cos((4 * Math.PI * n) / (M - 1));
}

function lowpassKernel(cutoff, fs, M) {
  const h = new Float64Array(M);
  const c = (M - 1) / 2;
  const fc = cutoff / fs;
  for (let n = 0; n < M; n++) {
    const m = n - c;
    const s = m === 0 ? 2 * fc : Math.sin(2 * Math.PI * fc * m) / (Math.PI * m);
    h[n] = s * blackman(n, M);
  }
  return h;
}

function bandpassKernel(low, high, fs, M) {
  const a = lowpassKernel(high, fs, M);
  const b = lowpassKernel(low, fs, M);
  for (let i = 0; i < M; i++) a[i] -= b[i];
  return a;
}

function hilbertKernel(M) {
  const h = new Float64Array(M);
  const c = (M - 1) / 2;
  for (let n = 0; n < M; n++) {
    const m = n - c;
    h[n] = m % 2 !== 0 ? (2 / (Math.PI * m)) * blackman(n, M) : 0;
  }
  return h;
}

// Fast convolution (overlap-add). Output is delay-compensated for a
// symmetric (linear-phase) kernel, so it lines up sample for sample with x.
function fftConvolve(x, h, onProgress) {
  const M = h.length;
  const delay = (M - 1) >> 1;
  const L = 8192;
  const N = nextPow2(L + M - 1);
  const Hr = new Float64Array(N), Hi = new Float64Array(N);
  for (let i = 0; i < M; i++) Hr[i] = h[i];
  fft(Hr, Hi);
  const full = new Float32Array(x.length + M - 1);
  const br = new Float64Array(N), bi = new Float64Array(N);
  const blocks = Math.ceil(x.length / L);
  for (let b = 0; b < blocks; b++) {
    const start = b * L;
    br.fill(0); bi.fill(0);
    const end = Math.min(start + L, x.length);
    for (let i = start; i < end; i++) br[i - start] = x[i];
    fft(br, bi);
    for (let k = 0; k < N; k++) {
      const r = br[k] * Hr[k] - bi[k] * Hi[k];
      const im = br[k] * Hi[k] + bi[k] * Hr[k];
      br[k] = r; bi[k] = im;
    }
    fft(br, bi, true);
    const lim = Math.min(N, full.length - start);
    for (let i = 0; i < lim; i++) full[start + i] += br[i];
    if (onProgress && (b % 16 === 0 || b === blocks - 1)) onProgress((b + 1) / blocks);
  }
  return full.slice(delay, delay + x.length);
}

// ---------- the pipeline ----------

const DEFAULTS = {
  carrier: 17500,   // Hz. Speech is moved to carrier .. carrier + bandwidth
  bandwidth: 3500,  // Hz of speech kept (telephone-plus quality)
  lowCut: 150,      // Hz, removes rumble that would sit right on the carrier
  level: 0.35,      // output peak (about -9 dBFS)
  repeats: 1,
  gapSeconds: 0.4,
  fadeSeconds: 0.03,
};

// Gentle compression so quiet and loud words encode at a similar strength.
function compress(x, drive = 3) {
  const p = peak(x) || 1;
  const norm = Math.tanh(drive);
  const out = new Float32Array(x.length);
  for (let i = 0; i < x.length; i++) out[i] = Math.tanh((drive * x[i]) / p) / norm;
  return out;
}

function withRepeats(x, repeats, gapSamples) {
  if (repeats <= 1) return x;
  const total = x.length * repeats + gapSamples * (repeats - 1);
  const out = new Float32Array(total);
  let pos = 0;
  for (let r = 0; r < repeats; r++) {
    out.set(x, pos);
    pos += x.length + gapSamples;
  }
  return out;
}

function fade(x, samples) {
  const n = Math.min(samples, x.length >> 1);
  for (let i = 0; i < n; i++) {
    const g = i / n;
    x[i] *= g;
    x[x.length - 1 - i] *= g;
  }
  return x;
}

function encodeSilent(speech, inRate, options = {}, onProgress = () => {}) {
  const o = { ...DEFAULTS, ...options };
  const fs = OUTPUT_RATE;
  if (o.carrier + o.bandwidth > fs / 2 - 1000) throw new Error('Carrier plus bandwidth is too close to the top of the audio range.');

  onProgress('Preparing your voice', 0);
  let x = resampleLinear(speech, inRate, fs);
  x = compress(x);
  x = withRepeats(x, o.repeats, Math.round(o.gapSeconds * fs));

  onProgress('Cleaning up the speech band', 0.05);
  const bp = bandpassKernel(o.lowCut, o.bandwidth, fs, 1023);
  x = fftConvolve(x, bp, (p) => onProgress('Cleaning up the speech band', 0.05 + p * 0.4));
  scaleTo(x, 1);

  onProgress('Shifting it above hearing range', 0.45);
  const xh = fftConvolve(x, hilbertKernel(511), (p) => onProgress('Shifting it above hearing range', 0.45 + p * 0.4));

  const w = (2 * Math.PI * o.carrier) / fs;
  const y = new Float32Array(x.length);
  for (let n = 0; n < x.length; n++) {
    y[n] = x[n] * Math.cos(w * n) - xh[n] * Math.sin(w * n);
  }
  scaleTo(y, o.level);
  fade(y, Math.round(o.fadeSeconds * fs));

  onProgress('Checking the hidden signal', 0.9);
  const report = verifySilent(y, fs, o.carrier, o.bandwidth);
  onProgress('Done', 1);
  return { audio: y, rate: fs, report, options: o };
}

// Measures where the energy in the file actually sits.
function verifySilent(y, fs, carrier = DEFAULTS.carrier, bandwidth = DEFAULTS.bandwidth) {
  const N = 4096;
  const win = new Float64Array(N);
  for (let i = 0; i < N; i++) win[i] = 0.5 - 0.5 * Math.cos((2 * Math.PI * i) / (N - 1));
  const frames = Math.max(1, Math.floor((y.length - N) / N));
  const step = Math.max(1, Math.floor(frames / 400));
  const spectrum = new Float64Array(N / 2);
  const re = new Float64Array(N), im = new Float64Array(N);
  let used = 0;
  for (let f = 0; f < frames; f += step) {
    const off = f * N;
    for (let i = 0; i < N; i++) { re[i] = (y[off + i] || 0) * win[i]; im[i] = 0; }
    fft(re, im);
    for (let k = 0; k < N / 2; k++) spectrum[k] += re[k] * re[k] + im[k] * im[k];
    used++;
  }
  const binHz = fs / N;
  const lo = carrier - 500, hi = carrier + bandwidth + 500;
  let inBand = 0, audible = 0, total = 0;
  for (let k = 1; k < N / 2; k++) {
    const hz = k * binHz;
    const p = spectrum[k];
    total += p;
    if (hz >= lo && hz <= hi) inBand += p;
    if (hz >= 20 && hz <= 15000) audible += p;
  }
  const share = total > 0 ? inBand / total : 0;
  const leakDb = inBand > 0 ? 10 * Math.log10((audible + 1e-20) / inBand) : 0;
  const pk = peak(y);
  const pass = share > 0.95 && leakDb < -30 && pk > 0.01;
  // Down-sampled spectrum in dB for drawing (256 points, 0 .. fs/2)
  const draw = new Float32Array(256);
  const per = (N / 2) / 256;
  let maxDb = -Infinity;
  for (let i = 0; i < 256; i++) {
    let s = 0;
    for (let k = Math.floor(i * per); k < Math.floor((i + 1) * per); k++) s += spectrum[k];
    draw[i] = 10 * Math.log10(s / Math.max(1, used) + 1e-20);
    if (draw[i] > maxDb) maxDb = draw[i];
  }
  for (let i = 0; i < 256; i++) draw[i] -= maxDb;
  return {
    pass,
    bandShare: share,
    audibleLeakDb: leakDb,
    peakDb: 20 * Math.log10(pk || 1e-9),
    seconds: y.length / fs,
    bandLowHz: lo,
    bandHighHz: hi,
    spectrumDb: draw,
    nyquistHz: fs / 2,
  };
}

// Shifts the hidden layer back down so it can be heard (Reveal).
function revealSilent(y, fs, carrier = DEFAULTS.carrier, bandwidth = DEFAULTS.bandwidth, onProgress) {
  const w = (2 * Math.PI * carrier) / fs;
  const m = new Float32Array(y.length);
  for (let n = 0; n < y.length; n++) m[n] = 2 * y[n] * Math.cos(w * n);
  const out = fftConvolve(m, lowpassKernel(bandwidth + 100, fs, 1023), onProgress);
  return scaleTo(out, 0.8);
}

// ---------- WAV ----------

function encodeWav(samples, rate) {
  const n = samples.length;
  const buf = new ArrayBuffer(44 + n * 2);
  const v = new DataView(buf);
  const str = (o, s) => { for (let i = 0; i < s.length; i++) v.setUint8(o + i, s.charCodeAt(i)); };
  str(0, 'RIFF'); v.setUint32(4, 36 + n * 2, true); str(8, 'WAVE');
  str(12, 'fmt '); v.setUint32(16, 16, true); v.setUint16(20, 1, true); v.setUint16(22, 1, true);
  v.setUint32(24, rate, true); v.setUint32(28, rate * 2, true); v.setUint16(32, 2, true); v.setUint16(34, 16, true);
  str(36, 'data'); v.setUint32(40, n * 2, true);
  let o = 44;
  for (let i = 0; i < n; i++, o += 2) {
    const s = Math.max(-1, Math.min(1, samples[i]));
    v.setInt16(o, s < 0 ? s * 0x8000 : s * 0x7fff, true);
  }
  return buf;
}

function decodeWav(buf) {
  const v = new DataView(buf);
  let pos = 12, rate = 48000, data = null, channels = 1, bits = 16;
  while (pos + 8 <= v.byteLength) {
    const id = String.fromCharCode(v.getUint8(pos), v.getUint8(pos + 1), v.getUint8(pos + 2), v.getUint8(pos + 3));
    const size = v.getUint32(pos + 4, true);
    if (id === 'fmt ') { channels = v.getUint16(pos + 10, true); rate = v.getUint32(pos + 12, true); bits = v.getUint16(pos + 22, true); }
    if (id === 'data') data = { off: pos + 8, size };
    pos += 8 + size + (size & 1);
  }
  if (!data || bits !== 16) throw new Error('Only 16-bit WAV files are supported.');
  const frames = data.size / 2 / channels;
  const out = new Float32Array(frames);
  for (let i = 0; i < frames; i++) out[i] = v.getInt16(data.off + i * 2 * channels, true) / 0x8000;
  return { samples: out, rate };
}

  }

});

async function getTTS() {
  if (state.tts) return state.tts;
  setProgress(0, 'Downloading the voice model (one time only)...');
  const { KokoroTTS } = await import(LIB.kokoro);
  const files = {};
  state.tts = await KokoroTTS.from_pretrained(LIB.kokoroModel, {
    dtype: 'q8',
    device: 'wasm',
    progress_callback: (p) => {
      if (p.status === 'progress' && p.total) {
        files[p.file] = [p.loaded, p.total];
        const [l, t] = Object.values(files).reduce((a, [x, y]) => [a[0] + x, a[1] + y], [0, 0]);
        setProgress(l / t, `Downloading the voice model: ${Math.round((l / t) * 100)}% (one time only)`);
      }
    },
  });
  return state.tts;
}

function splitText(text, maxLen = 260) {
  const sentences = text.replace(/\s+/g, ' ').match(/[^.!?]+[.!?]+["')\]]*|\S[^.!?]*$/g) || [];
  const chunks = [];
  let cur = '';
  for (const s of sentences) {
    if ((cur + ' ' + s).length > maxLen && cur) { chunks.push(cur.trim()); cur = s; } else cur += ' ' + s;
  }
  if (cur.trim()) chunks.push(cur.trim());
  return chunks.flatMap((c) => (c.length <= maxLen * 1.6 ? [c] : c.match(new RegExp(`.{1,${maxLen}}(\\s|$)`, 'g')) || [c]));
}

async function synthesize(text, voice, onProgress) {
  const tts = await getTTS();
  const chunks = splitText(text);
  const pieces = [];
  let rate = 24000;
  const started = Date.now();
  for (let i = 0; i < chunks.length; i++) {
    const out = await tts.generate(chunks[i], { voice });
    rate = out.sampling_rate;
    pieces.push(out.audio, new Float32Array(Math.round(rate * 0.25)));
    const done = (i + 1) / chunks.length;
    const secsLeft = ((Date.now() - started) / done) * (1 - done) / 1000;
    onProgress(done, `Reading your words aloud: part ${i + 1} of ${chunks.length}${i > 0 ? `, about ${Math.ceil(secsLeft / 60)} min left` : ''}`);
  }
  const total = pieces.reduce((a, p) => a + p.length, 0);
  const samples = new Float32Array(total);
  let pos = 0;
  for (const p of pieces) { samples.set(p, pos); pos += p.length; }
  return { samples, rate };
}

function playSamples(samples, rate) {
  const c = ctx();
  const buf = c.createBuffer(1, samples.length, rate);
  buf.copyToChannel(samples, 0);
  const src = c.createBufferSource();
  src.buffer = buf;
  src.connect(c.destination);
  src.start();
}

async function decodeToMono(arrayBuffer) {
  const ab = await ctx().decodeAudioData(arrayBuffer);
  const mono = new Float32Array(ab.length);
  for (let ch = 0; ch < ab.numberOfChannels; ch++) {
    const d = ab.getChannelData(ch);
    for (let i = 0; i < d.length; i++) mono[i] += d[i] / ab.numberOfChannels;
  }
  return { samples: mono, rate: ab.sampleRate };
}

const worker = new Worker(URL.createObjectURL(new Blob([QS_WORKER_SRC], { type: 'text/javascript' })));
let jobId = 0;
const jobs = new Map();
worker.onmessage = (e) => {
  const job = jobs.get(e.data.id);
  if (!job) return;
  if (e.data.type === 'progress') job.onProgress(e.data.p, e.data.stage);
  else if (e.data.type === 'done') { jobs.delete(e.data.id); job.resolve(e.data); }
  else if (e.data.type === 'error') { jobs.delete(e.data.id); job.reject(new Error(e.data.message)); }
};
function runJob(msg, transfer, onProgress) {
  return new Promise((resolve, reject) => {
    const id = ++jobId;
    jobs.set(id, { resolve, reject, onProgress });
    worker.postMessage({ ...msg, id }, transfer);
  });
}

function limitSeconds(sp) {
  if (isUnlocked()) return sp;
  const max = Math.round(FREE_SECONDS * sp.rate);
  return sp.samples.length > max ? { samples: sp.samples.slice(0, max), rate: sp.rate } : sp;
}

function drawSpectrum(r) {
  const cv = $('spectrum');
  const dpr = window.devicePixelRatio || 1;
  const W = cv.clientWidth || 680, H = cv.clientHeight || 150;
  cv.width = Math.round(W * dpr); cv.height = Math.round(H * dpr);
  const g = cv.getContext('2d');
  g.setTransform(dpr, 0, 0, dpr, 0, 0);
  g.clearRect(0, 0, W, H);
  const xOf = (hz) => (hz / r.nyquistHz) * W;
  g.fillStyle = 'rgba(143,180,255,0.10)';
  g.fillRect(xOf(r.bandLowHz), 0, xOf(r.bandHighHz) - xOf(r.bandLowHz), H);
  g.fillStyle = 'rgba(255,255,255,0.04)';
  g.fillRect(0, 0, xOf(15000), H);
  const grad = g.createLinearGradient(0, 0, W, 0);
  grad.addColorStop(0, '#8fb4ff'); grad.addColorStop(1, '#b79cff');
  g.strokeStyle = grad; g.lineWidth = 2; g.beginPath();
  const n = r.spectrumDb.length;
  for (let i = 0; i < n; i++) {
    const db = Math.max(-90, r.spectrumDb[i]);
    const x = (i / (n - 1)) * W, y = H - 18 - ((db + 90) / 90) * (H - 30);
    i ? g.lineTo(x, y) : g.moveTo(x, y);
  }
  g.stroke();
  g.fillStyle = '#9aa6c8'; g.font = '12px sans-serif';
  g.fillText('What people hear', 8, H - 4);
  g.fillText('Your hidden words', Math.min(W - 120, xOf(r.bandLowHz) + 4), 14);
  [5000, 10000, 15000, 20000].forEach((hz) => g.fillText(`${hz / 1000}k`, xOf(hz) - 8, H - 4 - 14));
}

$('spk').onclick = async () => {
  const tones = [15000, 17500, 19000, 20500, 21500];
  $('spk').disabled = true;
  $('tones').innerHTML = tones.map((f) => `<div class="tone" id="t${f}">${(f / 1000).toFixed(1)} kHz<br><span class="muted">waiting</span></div>`).join('');
  $('spkresult').textContent = '';
  let stream;
  try {
    stream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false } });
    const c = ctx(); await c.resume();
    const mic = c.createMediaStreamSource(stream);
    const an = c.createAnalyser(); an.fftSize = 8192; an.smoothingTimeConstant = 0.2;
    mic.connect(an);
    const data = new Float32Array(an.frequencyBinCount);
    const binOf = (f) => Math.round(f / (c.sampleRate / an.fftSize));
    const level = (f) => { an.getFloatFrequencyData(data); const b = binOf(f); return Math.max(data[b - 1], data[b], data[b + 1]); };
    const wait = (ms) => new Promise((r) => setTimeout(r, ms));
    await wait(400);
    const results = [];
    for (const f of tones) {
      if (f >= c.sampleRate / 2) { results.push([f, false]); $(`t${f}`).lastElementChild.textContent = 'not supported'; continue; }
      let base = 0; for (let k = 0; k < 4; k++) { base += level(f) / 4; await wait(40); }
      const osc = c.createOscillator(); const gn = c.createGain();
      osc.frequency.value = f; gn.gain.value = 0.15; osc.connect(gn).connect(c.destination); osc.start();
      await wait(350);
      let on = -Infinity; for (let k = 0; k < 6; k++) { on = Math.max(on, level(f)); await wait(50); }
      osc.stop();
      const heard = on - base > 10;
      results.push([f, heard]);
      const el = $(`t${f}`).lastElementChild;
      el.textContent = heard ? 'detected' : 'not detected';
      el.className = heard ? 'pass' : 'fail';
      await wait(200);
    }
    const ok = results.filter(([f, h]) => h && f >= 17500).map(([f]) => f);
    $('spkresult').textContent = ok.length
      ? `Your setup produced tones up to ${(Math.max(...ok) / 1000).toFixed(1)} kHz. ${Math.max(...ok) >= 20500 ? 'Standard and Higher should both work.' : 'Use the Standard setting.'}`
      : 'No high tones were detected. Your speaker or microphone may cut them off. Try earbuds held to the mic, or a different speaker.';
  } catch {
    showModal('Microphone needed', 'The speaker check listens with your microphone. Allow access and try again.');
  }
  stream && stream.getTracks().forEach((t) => t.stop());
  $('spk').disabled = false;
};
